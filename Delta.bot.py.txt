import os
import time
import threading
import requests
import pandas as pd
import ta
import numpy as np
from http.server import HTTPServer, BaseHTTPRequestHandler

# Dummy HTTP server to satisfy Render's port scan requirement
class SimpleHTTPRequestHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_response(200)
        self.end_headers()
        self.wfile.write(b"Bot is Running Alive!")

def run_dummy_server():
    port = int(os.environ.get("PORT", 10000))
    server = HTTPServer(('0.0.0.0', port), SimpleHTTPRequestHandler)
    server.serve_forever()

# Start dummy server in background thread so Render port error goes away
threading.Thread(target=run_dummy_server, daemon=True).start()

# Config & Settings
BASE_URLS = [
    "https://api.india.delta.exchange",
    "https://api.delta.exchange"
]
SYMBOL = "ETHUSD_PERP"  # Set to ETHUSD_PERP for Ethereum Perpetual Futures

ATR_LEN = 14
VOL_MA_LEN = 20
EMA_100_LEN = 100

def get_historical_candles(symbol=SYMBOL, resolution="15", limit=200):
    end_time = int(time.time())
    start_time = end_time - (limit * 15 * 60)
    
    params = {
        "symbol": symbol,
        "resolution": resolution,
        "from": start_time,
        "to": end_time
    }

    for base_url in BASE_URLS:
        endpoint = f"{base_url}/v2/chart/history"
        try:
            res = requests.get(endpoint, params=params, timeout=10)
            if res.status_code == 200:
                data = res.json()
                candles_data = None
                if "result" in data:
                    if isinstance(data["result"], list):
                        candles_data = data["result"]
                    elif isinstance(data["result"], dict):
                        candles_data = data["result"].get("candles", data["result"])

                if candles_data and len(candles_data) > 0:
                    df = pd.DataFrame(candles_data)
                    time_col = 'time' if 'time' in df.columns else ('start_time' if 'start_time' in df.columns else None)
                    if time_col:
                        df['time'] = pd.to_datetime(df[time_col], unit='s' if df[time_col].iloc[0] < 1e11 else 'ms')
                        df = df.sort_values('time').reset_index(drop=True)
                        return df
        except Exception:
            continue

    return None

def run_strategy():
    df = get_historical_candles()
    if df is None or len(df) < 10:
        print("Fetching candles from Delta Exchange API...")
        return False

    # Standard 'ta' library calculations
    df['atr'] = ta.volatility.average_true_range(df['high'], df['low'], df['close'], window=ATR_LEN)
    df['volMa'] = ta.trend.sma_indicator(df['volume'], window=VOL_MA_LEN)
    df['ema100'] = ta.trend.ema_indicator(df['close'], window=EMA_100_LEN)

    curr = df.iloc[-1]
    print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] Active | Symbol: {SYMBOL} | Candle Time: {curr['time']} | Close Price: {curr['close']}")
    return True

if __name__ == "__main__":
    print("Combined Breakout Engine v6 - Delta Exchange Python Bot Running...")
    while True:
        try:
            run_strategy()
        except Exception as e:
            print(f"Error in execution loop: {e}")
        time.sleep(60)
