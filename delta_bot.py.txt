def generate_signature(method, endpoint, payload=""):
    timestamp = str(int(time.time()))
    signature_data = method + timestamp + endpoint + payload
    message = bytes(signature_data, 'utf-8')
    secret = bytes(API_SECRET, 'utf-8')
    hash_value = hmac.new(secret, message, hashlib.sha256).hexdigest()
    return hash_value, timestamp

def place_order(side, size=CONTRACT_SIZE):
    endpoint = "/v2/orders"
    url = f"{BASE_URL}{endpoint}"
    
    payload = {
        "product_symbol": SYMBOL,
        "size": size,
        "side": side,  # "buy" or "sell"
        "order_type": "market_order"
    }
    
    import json
    payload_str = json.dumps(payload)
    signature, timestamp = generate_signature("POST", endpoint, payload_str)
    
    headers = {
        "api-key": API_KEY,
        "signature": signature,
        "timestamp": timestamp,
        "Content-Type": "application/json"
    }
    
    try:
        response = requests.post(url, data=payload_str, headers=headers)
        if response.status_code == 200 or response.status_code == 201:
            print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] SUCCESS: {side.upper()} order placed successfully!")
            return response.json()
        else:
            print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] FAILED Order: {response.text}")
            return None
    except Exception as e:
        print(f"Order exception: {e}")
        return None

def get_historical_candles():
    end_time = int(time.time())
    start_time = end_time - (200 * 15 * 60)
    
    endpoint = f"{BASE_URL}/v2/chart/history"
    params = {
        "symbol": SYMBOL,
        "resolution": "15",
        "from": start_time,
        "to": end_time
    }

    try:
        res = requests.get(endpoint, params=params, timeout=10)
        if res.status_code == 200:
            data = res.json()
            candles_data = data.get("result", [])
            if isinstance(candles_data, dict):
                candles_data = candles_data.get("candles", [])
            
            if candles_data:
                df = pd.DataFrame(candles_data)
                time_col = 'time' if 'time' in df.columns else 'start_time'
                df['time'] = pd.to_datetime(df[time_col], unit='s' if df[time_col].iloc[0] < 1e11 else 'ms')
                df = df.sort_values('time').reset_index(drop=True)
                return df
    except Exception as e:
        print(f"Error fetching candles: {e}")
    return None
