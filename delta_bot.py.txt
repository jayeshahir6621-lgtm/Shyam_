# ==================== HELPER FUNCTIONS ====================
def get_historical_candles(symbol=SYMBOL, resolution=TIMEFRAME, limit=300):
    """Fetch candles from Delta Exchange API"""
    endpoint = f"{DELTA_BASE_URL}/v2/chart/history"
    
    # 15-min candles timestamp calculation (in seconds)
    end_time = int(time.time())
    start_time = end_time - (limit * 15 * 60)
    
    params = {
        "symbol": symbol,
        "resolution": str(resolution),
        "from": start_time,
        "to": end_time
    }

    res = requests.get(endpoint, params=params)

    if res.status_code != 200:
        print(f"API Error {res.status_code}: {res.text}")
        return None

    data = res.json()

    if "result" in data:
        candles_data =data["result"].get("candles",data["result"]) if isinstance(data["result"],dict) else data["result"]
        if candles_data:
            df = pd.DataFrame(candles_data)
            df['time'] = pd.to_datetime(df['time'], unit='s' if df['time'].iloc[0] < 1e11 else 'ms')
            df = df.sort_values('time').reset_index(drop=True)
            return df

    return None


def calculate_pivot_high_low(df, length=12):
    """Calculate Pivot High and Pivot Low"""
    df['ph'] = np.nan
    df['pl'] = np.nan
    
    for i in range(length, len(df) - length):
        high_window = df['high'].iloc[i - length : i + length + 1]
        if df['high'].iloc[i] == high_window.max():
            df.at[i, 'ph'] = df['high'].iloc[i]
            
        low_window = df['low'].iloc[i - length : i + length + 1]
        if df['low'].iloc[i] == low_window.min():
            df.at[i, 'pl'] = df['low'].iloc[i]
            
    df['resLevel'] = df['ph'].ffill()
    df['supLevel'] = df['pl'].ffill()
    return df

def run_strategy():
    """Execute strategy checks and generate trade signals"""
    df = get_historical_candles()
    if df is None or len(df) < 150:
        print("Not enough candle data fetched.")
        return None

    # Indicators
    df['atr'] = ta.atr(df['high'], df['low'], df['close'], length=ATR_LEN)
    df['candleRange'] = df['high'] - df['low']
    df['isStrongCandle'] = df['candleRange'] >= (df['atr'] * ATR_MULT)

    df['volMa'] = ta.sma(df['volume'], length=VOL_MA_LEN)
    df['isStandardVol'] = df['volume'] >= (df['volMa'] * VOL_MULT)
    df['isHighVolume'] = df['isStandardVol'] if USE_VOLUME_FILTER else True

    df['ema100'] = ta.ema(df['close'], length=EMA_100_LEN)
    df['isEma100Bullish'] = df['close'] > df['ema100'] if USE_EMA_100 else True
    df['isEma100Bearish'] = df['close'] < df['ema100'] if USE_EMA_100 else True

    # ADX Calculation
    dmi = ta.adx(df['high'], df['low'], df['close'], length=ADX_LEN)
    df['adx'] = dmi[f'ADX_{ADX_LEN}']
    df['isAdxValid'] = df['adx'] >= ADX_THRESHOLD if USE_ADX else True

    # Bollinger Bands Squeeze Calculation
    bb = ta.bbands(df['close'], length=BB_LEN, std=BB_MULT)
    df['bbMiddle'] = bb[f'BBM_{BB_LEN}_{BB_MULT}']
    df['bbUpper'] = bb[f'BBU_{BB_LEN}_{BB_MULT}']
    df['bbLower'] = bb[f'BBL_{BB_LEN}_{BB_MULT}']
    df['bbWidthPercent'] = ((df['bbUpper'] - df['bbLower']) / df['bbMiddle']) * 100
    df['isSqueezed'] = (df['bbWidthPercent'].shift(1) <= SQUEEZE_THRESHOLD) | (df['bbWidthPercent'].shift(2) <= SQUEEZE_THRESHOLD) if USE_SQUEEZE else True

    # Pivot High / Low
    df = calculate_pivot_high_low(df, length=PIVOT_LEN)

    # Latest completed bar
    curr = df.iloc[-2]
    prev = df.iloc[-3]

    res_level = curr['resLevel']
    sup_level = curr['supLevel']
    buffer_dist = (curr['atr'] * BREAKOUT_BUFFER_MULT) if USE_BREAKOUT_BUFFER else 0.0

    # Signals
    bullish_break = (pd.notna(res_level) and 
                     curr['close'] > (res_level + buffer_dist) and 
                     prev['close'] <= res_level and 
                     curr['isHighVolume'] and 
                     curr['isStrongCandle'] and 
                     curr['isEma100Bullish'] and 
                     curr['isAdxValid'] and 
                     curr['isSqueezed'])

    bearish_break = (pd.notna(sup_level) and 
                     curr['close'] < (sup_level - buffer_dist) and 
                     prev['close'] >= sup_level and 
                     curr['isHighVolume'] and 
                     curr['isStrongCandle'] and 
                     curr['isEma100Bearish'] and 
                     curr['isAdxValid'] and 
                     curr['isSqueezed'])

    if bullish_break:
        entry = curr['close']
        sl = entry - (curr['atr'] * SL_ATR_MULT)
        tp = entry + ((entry - sl) * RR_RATIO)
        return {"side": "buy", "entry": entry, "sl": sl, "tp": tp}

    if bearish_break:
        entry = curr['close']
        sl = entry + (curr['atr'] * SL_ATR_MULT)
        tp = entry - ((sl - entry) * RR_RATIO)
        return {"side": "sell", "entry": entry, "sl": sl, "tp": tp}

    return None
